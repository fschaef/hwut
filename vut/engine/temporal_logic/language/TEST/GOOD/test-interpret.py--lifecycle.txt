
--- lifecycle (run-lifecycle.vut) ---
run    plant
wind   f : script
make   s : station
enter  s.polling
recur  poll every 0.5s
enter  s.master
wire   f --[cmd]--> s
feed   f start
event  s.cmd start
tick   0.5s poll
emit   s.* poll
event  s.. poll
tick   1.0s poll
emit   s.* poll
event  s.. poll
feed   f mute
event  s.cmd mute
cancel poller
feed   f start
event  s.cmd start
feed   f stop
event  s.cmd stop
leave  s.polling
