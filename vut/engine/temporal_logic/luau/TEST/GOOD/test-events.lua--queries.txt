
--- existence and counting ---
empty()            = no
any()              = yes
has(ip=a)          = yes
has(ip=z)          = no
list(ip=a) count   = 2

--- all / none ---
all(ip=a)          = no
none(ip=z)         = yes

--- temporal: last / since ---
last() time        = 3
last(ip=a) time    = 3
since(now=10,ip=a) = 7

--- comparator condition: time > 1 ---
list(time>1) count = 2
