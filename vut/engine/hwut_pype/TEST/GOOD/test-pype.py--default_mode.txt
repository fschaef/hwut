
--- default mode starts; leaving it is one-way ---
SCRIPT: {
    on: <entry> => {
        print("default begin")
    }
    on: "tick" => {
        print("default tick")
    }
    on: "leave" => goto NAMED;
    on: <exit> => {
        print("default end")
    }
    NAMED/on: "tick" => {
        print("NAMED tick")
    }
    on: <else> => ignore;
    NAMED/on: <else> => ignore;
}
INPUT: {
    tick
    leave
    tick
}
OUTPUT: {
default begin
default tick
default end
NAMED tick
}

}
