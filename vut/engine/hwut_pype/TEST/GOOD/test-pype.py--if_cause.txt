--- condition reads the shared namespace ---
SCRIPT: {
    M/on: <entry> => {
        budget = 2
    }
    M/on: "spend" => {
        budget -= 1
        print("spent; budget", budget)
    }
    M/on: if ( budget <= 0 ) => {
        print("budget exhausted at %r" % pype.line())
    }
    M/on: <else> => ignore;
}
INPUT: {
    spend
    idle
    spend
    idle
}
OUTPUT: {
spent; budget 1
spent; budget 0
budget exhausted at 'idle'
}
