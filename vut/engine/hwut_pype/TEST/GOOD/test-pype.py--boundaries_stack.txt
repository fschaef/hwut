
--- <bof> and <eof> fire once at the file boundaries ---
SCRIPT: {
    on: <bof> => {
        print("<bof> at line_n", pype.line_n())
    }
    on: "x" => {
        print("x at line_n", pype.line_n())
    }
    on: <else> => ignore;
    on: <eof> => {
        print("<eof> at line_n", pype.line_n(),
              "; pype.time() >= 0:", pype.time() >= 0.0)
    }
}
INPUT: {
    x
    noise
    x
}
OUTPUT: {
<bof> at line_n 0
x at line_n 1
x at line_n 3
<eof> at line_n 3 ; pype.time() >= 0: True
}

--- PUSH suspends, POP returns; <entry>/<exit> fire on both ---
SCRIPT: {
    on: <entry> => {
        print("default <entry>")
    }
    on: "dig" => push DETAIL;
    on: "x" => {
        print("default x")
    }
    on: <else> => ignore;
    DETAIL/on: <entry> => {
        print("  DETAIL <entry>")
    }
    DETAIL/on: "up" => pop;
    DETAIL/on: "x" => {
        print("  DETAIL x")
    }
    DETAIL/on: <else> => ignore;
    DETAIL/on: <exit> => {
        print("  DETAIL <exit>")
    }
}
INPUT: {
    x
    dig
    x
    up
    x
}
OUTPUT: {
default <entry>
default x
  DETAIL <entry>
  DETAIL x
  DETAIL <exit>
default <entry>
default x
}

--- POP may return into the nameless default mode ---
SCRIPT: {
    on: "call" => push SUB;
    on: "x" => {
        print("default x")
    }
    on: <else> => ignore;
    SUB/on: "ret" => pop;
    SUB/on: <else> => ignore;
}
INPUT: {
    call
    ret
    x
}
OUTPUT: {
default x
}

--- pype.bindings(): the current line's pattern bindings ---
SCRIPT: {
    M/on: "p" +( i : <x[i] = int> ) => {
        print("bindings:", dict(sorted(pype.bindings().items())))
    }
    M/on: if ( True ) => {
        print("if fire, bindings:", pype.bindings())
    }
    M/on: <else> => ignore;
}
INPUT: {
    p 4 5
    anything else
}
OUTPUT: {
bindings: {'i': 2, 'x': {0: 4, 1: 5}}
if fire, bindings: {}
}
