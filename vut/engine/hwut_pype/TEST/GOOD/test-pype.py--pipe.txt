
--- two input files read in order as one stream ---
EXIT: 0
    l1 x
    l3 x

--- trace positions name the input file, per-file line count ---
    filter.pype: mode active: <default>
    in-a.txt:1: input: 'l1 x'
    filter.pype:1: fire: <default>/on: "x"
    filter.pype:1: effect: flush ;
    in-a.txt:2: input: 'l2'
    filter.pype:2: fire: <default>/on: <else>
    in-b.txt:1: input: 'l3 x'
    filter.pype:1: fire: <default>/on: "x"
    filter.pype:1: effect: flush ;
SCRIPT: 'tmp-filter.pype' (she-bang: hwut.pype) {
    SCAN/on: "PASS" <n = int> ":" <f = "*.txt"> => {
        print("ok %d %s" % (n, f))
    }
    SCAN/on: <exit> => {
        print("done")
    }
    SCAN/on: <else> => ignore;
}
EXIT: 0
STDOUT: {
    ok 1 a.txt
    ok 2 b.txt
    done
}
