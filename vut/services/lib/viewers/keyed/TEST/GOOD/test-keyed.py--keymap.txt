P-20/P-21/P-22: the table is the keymap

-- the help screen IS the table
     PANE
       <space>       mark a range here
       <enter>       copy marked range to GOOD
       a             copy this region to GOOD
       A             copy the whole OUTPUT to GOOD
       d             remove from GOOD: the marked lines, or the region here
       j, <down>     down
       k, <up>       up
       <c-d>         a screen down
       <c-u>         a screen up
       h, <left>     scroll left
       l, <right>    scroll right
       w             next element that can vary or differs; the foot says what it is
       b             previous such element
       /             search down
       ?             search up
       <number>g     go to that OUTPUT line
     
     GLOBAL
       <tab>         switch pane
       u             undo
       r             redo
       R             reset: GOOD as it stood when the screen opened
       z             re-align: ask compare again
       t             the tolerance report: analogies, patterns, constraints
       e             edit the GOOD in $EDITOR
       <F1>          this table
       q             done: GOOD as it stands is written
       <c-c>         cancel: nothing is written

-- no key is bound twice
     keyed   : none
     fallback: none

-- the fallback speaks the SAME acts
     t    -> TAKE_ALL
     e    -> EDIT
     c    -> DONE
     q    -> CANCEL

-- an unbound key means nothing, and that is not an error
     act_of('Z') = None
<hwut-end>
