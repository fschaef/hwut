    }
}

    # starts here
    value # trailing
}

}

--- eol demands the pattern reaches line end ---

    M/on: "done" <eol> => {

    }
}

    done deal
}

no: 'done deal'
}

--- repeat: one or more; counter and indexed dictionary bindings ---
SCRIPT: {
    M/on: "sum" +( i : <x[i] = number> ";" ) "end" => {
        print("n =", i, "values =", [x[k] for k in sorted(x)],
              "sum =", sum(x.values()))
    }
    M/on: <else> => {
        print("rejected: %r" % pype.line())
    }
}
INPUT: {
    sum 1.5 ; 2.5 ; 3.5 ; end
    sum end
}
OUTPUT: {
n = 3 values = [1.5, 2.5, 3.5] sum = 7.5
rejected: 'sum end'
}

--- offset expressions in the index brackets ---
SCRIPT: {
    M/on: "pair" +( i : <a[i + 2] = int> <b[i - 1] = int> ) => {
        print("i =", i, "a =", dict(sorted(a.items())),
              "b =", dict(sorted(b.items())))
    }
    M/on: <else> => ignore;
}
INPUT: {
    pair 10 11 20 21
}
OUTPUT: {
i = 2 a = {2: 10, 3: 20} b = {-1: 11, 0: 21}
}
